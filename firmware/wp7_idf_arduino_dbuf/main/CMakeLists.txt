# The approved WP7 welcome screen is a single, read-only RGB565 raster.  Keep
# it in flash rather than allocating a second framebuffer or a PSRAM staging
# surface.  The C++ renderer copies it once into the existing Arduino_GFX
# framebuffer before the first visible presentation.
set(PP_WP7_WELCOME_RASTER
    "${PP_FIRMWARE_ROOT}/assets/wp7_welcome_static_rgb565.bin")
file(SIZE "${PP_WP7_WELCOME_RASTER}" PP_WP7_WELCOME_RASTER_BYTES)
if(NOT PP_WP7_WELCOME_RASTER_BYTES EQUAL 768000)
  message(FATAL_ERROR
          "WP7 static welcome raster must be exactly 768000 bytes, got ${PP_WP7_WELCOME_RASTER_BYTES}")
endif()
target_add_binary_data(${COMPONENT_LIB} "${PP_WP7_WELCOME_RASTER}" BINARY)

# Gina's approved WP8 racing artwork is embedded byte-for-byte.  These opaque,
# native-size RGB565 assets are drawn directly into the existing framebuffer;
# they neither create a second framebuffer nor require a staging copy.
set(PP_WP7_APPROVED_ASSET_ROOT "${PP_FIRMWARE_ROOT}/assets/wp7_approved")
foreach(PP_WP7_APPROVED_ASSET
    lamp_red.rgb565
    lamp_green.rgb565
    lamp_off.rgb565
    lamp_yellow.rgb565
    header_trapezoid_surround.rgb565
    pp_header_logo.rgb565)
  target_add_binary_data(${COMPONENT_LIB}
    "${PP_WP7_APPROVED_ASSET_ROOT}/${PP_WP7_APPROVED_ASSET}" BINARY)
endforeach()

# Keep this target explicitly opt-in.  The default remains the historical
# disposable double-buffer diagnostic.  The XIP comparison build selects the
# already-proven single-framebuffer adapter through an environment-only switch
# and never changes the production PlatformIO environment.
target_compile_definitions(${COMPONENT_LIB} PRIVATE
  PP_PHASE8B_REAL_SD=1
  PP_PHASE8B_P1_TEST=1
  PP_WP7_PHYSICAL_RACING_TEST=1
  PP_UPDATE_TEST_KEY=1
  PP_MAX_ENTRIES=2
  PP_MAX_SSE_CLIENTS=4
  PP_WIFI_SCAN_INSTRUMENTATION=1
  ARDUINO_USB_CDC_ON_BOOT=0
  PP_WP7_STATIC_WELCOME_RASTER=1)
